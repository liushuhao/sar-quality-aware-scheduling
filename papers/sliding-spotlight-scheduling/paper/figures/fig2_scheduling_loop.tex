% fig-2: encoding-evaluation loop of the scheduler (schematic TikZ).
% Spec pointers: figure-plan-review C (must-show list + machine predicates),
% sss_core/evaluator.py:367-444 (the real loop this diagram mirrors).
% Predicates honoured here: the evaluator box contains NO AASR mention (the
% evaluator performs no AASR recheck -- admission lives on the generation side);
% the decode arrow is labelled as a measure projection (tau is a fraction of the
% feasible measure, not wall-clock time); the C2 pad is labelled (1+0.25) on the
% dwell; the anchor-seed arrow into the initial population is present; the sort
% node sits between decode and the C2/conflict chain; the evaluator box names
% the single objective -f_1 (f_q is a reported reading).
\resizebox{\textwidth}{!}{%
\begin{tikzpicture}[x=1mm,y=1mm,>=stealth,
  box/.style={draw,rounded corners=1.5,align=center,font=\footnotesize,
              inner sep=2,outer sep=0pt,minimum height=8,text width=31mm,
              execute at begin node={\baselineskip=3.4mm\hyphenpenalty=10000\exhyphenpenalty=10000\relax}},
  lab/.style={font=\footnotesize,align=center,outer sep=0pt,text width=44mm,
              execute at begin node={\baselineskip=3.4mm\hyphenpenalty=10000\exhyphenpenalty=10000\relax}}]
  % ------------------------------------------------ generation side
  \node[box,fill=blue!6] (gen) at (0,24)
    {scenario generation\\ \cite{kim2021optimal} platform, true ground-track windows};
  \node[box,fill=blue!6] (adm) at (37,24)
    {AASR admission\\ 11 evenly spaced in-window samples,\\ worst of the 11 vs threshold};
  \draw[->] (gen) -- (adm);
  \node[lab,fill=blue!6,text width=34mm] (gcal) at (74,24)
    {generation-side caliber: a served task is feasible w.r.t.\ the generator's admission};
  % ------------------------------------------------ genome
  \node[box] (chrom) at (0,0)
    {chromosome $x=(s,\tau)\in\{0,1\}^{N}\times[0,1]^{N}$\\
     $s$: selection bits; $\tau$: start fractions\\
     no mode gene (product requirement fixes $m_i$)};
  \node[box,fill=orange!12] (seed) at (-37,0)
    {greedy anchor seed\\ (bounded re-timing repair)};
  \draw[->] (seed) -- (chrom);
  \node[lab,anchor=south east,text width=34mm] at (-18,12) {seed $\to$ initial population};
  % ------------------------------------------------ decode
  \node[box] (dec) at (37,0) {decode: $\tau\to$ start time\\
    \emph{measure projection} onto the union\\ of feasible window intervals (nonlinear)};
  \draw[->] (chrom) -- (dec);
  % ------------------------------------------------ sort + evaluate
  \node[box] (sort) at (37,-21) {sort tasks by start time\\ (index order $\neq$ time order)};
  \node[box] (c2) at (74,-25) {C2 chain: for consecutive pair\\
    $(1+0.25)\,T_{\mathrm{dwell}}+$\\
    $t_{\mathrm{man}}+t_{\mathrm{settle}}\le$ gap\\
    pad multiplies the dwell, never the maneuver};
  \node[box,dashed] (c34) at (74,0) {C3/C4: budgets normalised\\
    against the library envelope\\ --- slack by construction ---};
  \node[box] (conf) at (74,-46) {conflict check: imaging intervals\\ (bare dwell) pairwise disjoint};
  \node[box,fill=green!6] (eval) at (0,-28) {evaluator\\ objective $-f_1$\\
    violation count from the chains at left};
  % f_q is a reported reading only: side note beside
  % the box keeps the box itself single-objective
  \node[font=\footnotesize,align=left,anchor=east] at (-16.5,-28)
    {$f_q$: reported reading,\\coverage the only objective};
  \draw[->] (dec) -- (sort);
  \draw[->] (sort) -- (c2);
  \draw[->] (c2) -- (conf);
  \draw[->,dashed] (c34) -- (c2);
  \draw[->] (adm) -- (18.5,24) -- (18.5,-28) -- (eval);
  % elbow exits c2 low on its west side: a plain (c2) |- (eval) starts at c2's
  % south border and runs back up through the box (eval's centre y=-28 is above
  % c2's south border), crossing the pad/dwell/maneuver lines of box text
  \draw[->] ([yshift=-8mm]c2.west) -- ++(-2,0) |- ([yshift=-3mm]eval.east);
  \draw[->] (conf) -- (0,-46) -- (eval);
  \node[lab,anchor=north,text width=44mm] at (-28,-36)
    {admitted windows only; the evaluator performs no AASR recheck};
  % objectives note
  \node[lab,anchor=north west,text width=104mm] at (-50,-47)
    {$f_1=\sum_{i\in S}p_i/\sum_i p_i$ (served priority fraction)\\
     $f_q=\frac{1}{|S|}\sum_{i\in S}f_{q,i}$ (mean per-task quality, non-normalised)};
\end{tikzpicture}}
